# 核心引擎：只有 PyYAML 一个硬依赖
#   core / rules / export 三层零三方依赖，任何环境都能跑
pip install log-ai-compressor

# ---- 以下都是可选能力，按需装 ----

# 本地 Web 界面（双击 start.bat 走这条）
pip install "log-ai-compressor[web]"

# MCP Server：让 Claude Code / Codex / mavis 等 Agent 调用本机日志
pip install "log-ai-compressor[mcp]"

# AI 解读（OpenAI 兼容端点 / 本地 Ollama 都走它）
pip install "log-ai-compressor[ai]"

# 全都要
pip install "log-ai-compressor[all]"

# ---- 源码开发 ----
pip install -e ".[all,dev]"

# ---- 说明 ----
# AI 解读是**可选**功能，不装 httpx 也能用：
#   聚类 / 根因判定 / 异常检测 / 报告导出全是本地算法，零出网、零成本。
